// internal/contracts/vault.go
type VaultService interface {
    // Payout via vault
    Payout(ctx context.Context, req *PayoutRequest) (*PayoutResponse, error)
    
    // Verificar limites
    GetTokenPolicy(ctx context.Context, token string) (*TokenPolicy, error)
    
    // Verificar status
    GetVaultStatus(ctx context.Context) (*VaultStatus, error)
    
    // Pausar/despausar (emergência)
    Pause(ctx context.Context) error
    Unpause(ctx context.Context) error
    
    // Adicionar/remover operadores
    AddOperator(ctx context.Context, address string) error
    RemoveOperator(ctx context.Context, address string) error
}

// internal/contracts/registry.go
type RegistryService interface {
    // Verificar delegate confiável
    IsDelegateTrusted(ctx context.Context, address string) (bool, error)
    
    // Registrar delegate
    RegisterDelegate(ctx context.Context, address string, codeHash string) error
    
    // Revogar delegate
    RevokeDelegate(ctx context.Context, address string, reason string) error
}
