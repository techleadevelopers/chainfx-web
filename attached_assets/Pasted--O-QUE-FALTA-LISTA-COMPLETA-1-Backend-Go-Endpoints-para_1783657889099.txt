// internal/contracts/vault.go
type VaultService interface {
    // Payout via vault
    Payout(ctx context.Context, req *PayoutRequest) (*PayoutResponse, error)
    
    // Verificar limites
    GetTokenPolicy(ctx context.Context, token string) (*TokenPolicy, error)
    
    // Verificar status
    GetVaultStatus(ctx context.Context) (*VaultStatus, error)
    
    // Pausar/despausar (emergência)
    Pause(ctx context.Context) error
    Unpause(ctx context.Context) error
    
    // Adicionar/remover operadores
    AddOperator(ctx context.Context, address string) error
    RemoveOperator(ctx context.Context, address string) error
}

// internal/contracts/registry.go
type RegistryService interface {
    // Verificar delegate confiável
    IsDelegateTrusted(ctx context.Context, address string) (bool, error)
    
    // Registrar delegate
    RegisterDelegate(ctx context.Context, address string, codeHash string) error
    
    // Revogar delegate
    RevokeDelegate(ctx context.Context, address string, reason string) error
}

-- Tabelas para mobile
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(20) UNIQUE,
    full_name VARCHAR(255),
    password_hash VARCHAR(255) NOT NULL,
    wallet_address VARCHAR(42) UNIQUE,
    pix_key VARCHAR(255),
    kyc_status VARCHAR(20) DEFAULT 'pending',
    kyc_documents JSONB,
    pin_hash VARCHAR(255),
    biometry_enabled BOOLEAN DEFAULT false,
    two_factor_enabled BOOLEAN DEFAULT false,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE devices (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    device_name VARCHAR(255),
    device_type VARCHAR(50),
    fcm_token VARCHAR(255),
    apns_token VARCHAR(255),
    last_active TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE dca_strategies (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    token_symbol VARCHAR(10) NOT NULL,
    amount_brl DECIMAL(18,2) NOT NULL,
    frequency VARCHAR(20) NOT NULL, -- daily, weekly, monthly
    active BOOLEAN DEFAULT true,
    total_invested DECIMAL(18,2) DEFAULT 0,
    total_tokens DECIMAL(36,18) DEFAULT 0,
    next_execution TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    title VARCHAR(255) NOT NULL,
    body TEXT,
    type VARCHAR(50),
    read BOOLEAN DEFAULT false,
    data JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE settings (
    user_id UUID PRIMARY KEY REFERENCES users(id),
    dark_mode BOOLEAN DEFAULT true,
    language VARCHAR(10) DEFAULT 'pt-BR',
    currency VARCHAR(10) DEFAULT 'BRL',
    notifications_enabled BOOLEAN DEFAULT true,
    daily_limit DECIMAL(18,2) DEFAULT 10000
);
